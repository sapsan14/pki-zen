% PKI-ZEN XeLaTeX template.
%
% Fonts are chosen for CI reliability, not typographic elegance:
% - Noto Serif has Latin, Cyrillic, and all Estonian diacritics (Š Ž Õ Ä Ö Ü).
% - Noto Sans for headings.
% - DejaVu Sans Mono is always available via fonts-dejavu on Ubuntu.

\documentclass[12pt,oneside]{book}

\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{$babel-lang$}
\setotherlanguages{russian,estonian,english}

% Match the web/EPUB edition's typography: a humanist sans for body
% and headings. Noto Sans covers Latin, Cyrillic, and the Estonian
% extended set in a single family, which lets us avoid font-substitution
% shimmer between scripts. DejaVu Sans Mono stays as the code face —
% it's installed in the build image by default and reads cleanly next
% to Noto Sans.
\setmainfont{Noto Sans}[Ligatures={TeX,Common}]
\setsansfont{Noto Sans}
\setmonofont{DejaVu Sans Mono}[Scale=0.88]

% Noto Serif does not contain every Unicode glyph — in particular ∞
% (U+221E), which we use as the "closing verse" probability marker. Map
% it (and a handful of other glyphs we occasionally hit) to explicit
% math-mode commands that always render.
\usepackage{newunicodechar}
\newunicodechar{∞}{\ensuremath{\infty}}
\newunicodechar{⟨}{\ensuremath{\langle}}
\newunicodechar{⟩}{\ensuremath{\rangle}}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{←}{\ensuremath{\leftarrow}}

\usepackage{geometry}
\geometry{paperwidth=152mm, paperheight=229mm, margin=18mm,
          headheight=15pt, headsep=10mm, footskip=24pt}

\usepackage{microtype}
\usepackage{hyperref}
\hypersetup{colorlinks=true, linkcolor=black, urlcolor=black, citecolor=black}

\usepackage{amsmath}
\usepackage{enumitem}
\setlist{nosep}

% ----- Paragraph rhythm ------------------------------------------------
% The book reads better with flush-left paragraphs (no first-line
% indent) and a small skip between them. Matches the EPUB rendering.
% The generous `plus 1\baselineskip` stretch lets \flushbottom (below)
% distribute extra glue *between* verses on partially-filled pages,
% so the body block ends up vertically balanced on the page instead of
% collapsing to the top with a yawning gap underneath.
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5\baselineskip plus 1\baselineskip}

% Never orphan the first line of a paragraph at the bottom of a page,
% never widow the last line at the top of the next page.
\clubpenalty=10000
\widowpenalty=10000
% \flushbottom keeps the body text vertically balanced on every page
% by stretching the parskip glue between verses, instead of leaving a
% large empty band at the bottom of partially-filled pages.
\flushbottom

% ----- Code blocks -----------------------------------------------------
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines,breakanywhere,commandchars=\\\{\},fontsize=\small
}

% PDFs build with --no-highlight, so Pandoc does not emit Shaded / Tok
% macros. The Highlighting Verbatim env above handles plain mono code.

% Pandoc-injected helper commands our custom template must define.
% \providecommand so any future Pandoc that starts defining them wins.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{#1}

% ----- Page headers and footers ----------------------------------------
% Default LaTeX book class shows page numbers in inconsistent spots
% (top-right on some, nothing on others). Override with fancyhdr so
% every page has the chapter title centered at the top (auto-wrapping
% to two lines if long) and the page number centered at the bottom.
\usepackage{fancyhdr}
\usepackage{etoolbox}          % for \pretocmd on sectioning commands

\pagestyle{fancy}
\fancyhf{}                     % clear everything
\fancyhead[C]{\small\scshape\nouppercase{\leftmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% 'plain' pagestyle (used by \chapter) should match: centered page
% number at the bottom, no header.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% Allow a long chapter title to break across two lines in the header.
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}

% ----- Chapter titles --------------------------------------------------
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\bfseries\filcenter}
  {\fontsize{14pt}{16pt}\selectfont\textsc{\chaptertitlename\ \thechapter}}
  {12pt}
  {\fontsize{22pt}{26pt}\selectfont}
\titlespacing*{\chapter}{0pt}{20pt}{32pt}

% ----- Silence LaTeX's autogenerated section numbering -----------------
% secnumdepth = 0: number the chapter counter (so \thechapter still
% steps to 1, 2, 3, … and our titleformat prints "CHAPTER 1" in the
% body) but suppress numbers on every deeper level (section /
% subsection / …).
\setcounter{secnumdepth}{0}

% ...and in the TOC, strip the chapter number from the entry so the
% contents reads "Book I. Title ..." without a leading "1" that
% overlaps the 'B' of 'Book'. Done by patching \@chapter to drop the
% \numberline{\thechapter} it usually prepends to \addcontentsline.
\usepackage{etoolbox}
\makeatletter
\patchcmd{\@chapter}{\protect\numberline{\thechapter}#1}{#1}{}{}
\makeatother

% ----- Single-page Table of Contents -----------------------------------
% Chapters appear in the TOC with titles, no auto-numbers. Verses are
% *not* listed in the TOC — tocdepth = 0 keeps it to main book headings
% only, which is all a reader needs to navigate a paginated book.
\setcounter{tocdepth}{0}

% The global \parskip (0.5 baselineskip) inflates TOC lines enough to
% push the 13th entry onto a second page. Use tocloft to collapse the
% inter-chapter gap, and zero \parskip locally inside \tableofcontents,
% so the whole TOC fits on one page harmoniously.
\usepackage{tocloft}
% 7pt of space before each chapter line gives the entries room to
% breathe while still fitting all 13 of them on a single page.
\setlength{\cftbeforechapskip}{7pt}
\let\oldtableofcontents\tableofcontents
\renewcommand{\tableofcontents}{%
  \begingroup
    \setlength{\parskip}{0pt}%
    \oldtableofcontents
  \endgroup
}

% ----- Keep verses together --------------------------------------------
% Each ## v-heading becomes \section (documentclass=book maps # to
% \chapter, ## to \section). Demand 18 lines of runway below the
% heading; if less, advance to the next page. Covers the longest
% verses in the book (v9.1, v8.1 around 10–14 visual lines with their
% probability signature) so the whole verse stays on one page.
\usepackage{needspace}
\pretocmd{\section}{\Needspace*{18\baselineskip}}{}{}

% ----- Make verbatim blocks narrower -----------------------------------
% The colophon carries a 64-char SHA-256 fingerprint inside a fenced
% code block. At the default 12pt monospace size it overflows the 6-in
% text column. Render all verbatim blocks in \small so long fingerprints
% and commands fit within the page.
\AtBeginEnvironment{verbatim}{\small}

% ----- Closing probability line ---------------------------------------
% Each verse ends with a '*Probability: X*' line. The build script
% rewrites those in the stamped Markdown tree to \prob{...} so they
% pass cleanly through Pandoc's raw_tex extension (bare {...} groups
% are not recognised and would leak literal braces into the PDF).
\newcommand{\prob}[1]{{\footnotesize\itshape #1}}

% ----- Include SVG cover and back cover as PDF pages -------------------
% If dist/cover.pdf and dist/back-cover.pdf exist (the build script
% rasterises publish/covers/*.svg to PDF via rsvg-convert), they get
% included as first and last pages so the PDF is a "book with covers"
% out of the box.
\usepackage{pdfpages}

% Some metadata placeholders Pandoc may emit.
$if(title)$
\title{$title$}
$endif$
$if(author)$
\author{$for(author)$$author$$sep$ \and $endfor$}
$endif$
$if(date)$
\date{$date$}
$endif$

\begin{document}

% Front cover. First thing in the PDF, no page number.
$if(cover-pdf)$
\thispagestyle{empty}
\includepdf[pages=1,fitpaper=true]{$cover-pdf$}
$endif$

$if(title)$
\maketitle
$endif$

$if(toc)$
\tableofcontents
\clearpage
$endif$

$body$

% Back cover. Last thing in the PDF.
$if(back-cover-pdf)$
\thispagestyle{empty}
\clearpage
\includepdf[pages=1,fitpaper=true]{$back-cover-pdf$}
$endif$

\end{document}
