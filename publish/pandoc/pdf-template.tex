% PKI-ZEN XeLaTeX template.
%
% Fonts are chosen for CI reliability, not typographic elegance:
% - Noto Serif has Latin, Cyrillic, and all Estonian diacritics (Š Ž Õ Ä Ö Ü).
% - Noto Sans for headings.
% - DejaVu Sans Mono is always available via fonts-dejavu on Ubuntu.

\documentclass[12pt,oneside]{book}

\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{$babel-lang$}
\setotherlanguages{russian,estonian,english}

\setmainfont{Noto Serif}[Ligatures={TeX,Common}]
\setsansfont{Noto Sans}
\setmonofont{DejaVu Sans Mono}[Scale=0.88]

% Noto Serif does not contain every Unicode glyph — in particular ∞
% (U+221E), which we use as the "closing verse" probability marker. Map
% it (and a handful of other glyphs we occasionally hit) to explicit
% math-mode commands that always render.
\usepackage{newunicodechar}
\newunicodechar{∞}{\ensuremath{\infty}}
\newunicodechar{⟨}{\ensuremath{\langle}}
\newunicodechar{⟩}{\ensuremath{\rangle}}

\usepackage{geometry}
\geometry{paperwidth=152mm, paperheight=229mm, margin=18mm,
          headheight=15pt, footskip=24pt}

\usepackage{microtype}
\usepackage{hyperref}
\hypersetup{colorlinks=true, linkcolor=black, urlcolor=black, citecolor=black}

\usepackage{amsmath}
\usepackage{enumitem}
\setlist{nosep}

% ----- Paragraph rhythm ------------------------------------------------
% The book reads better with flush-left paragraphs (no first-line
% indent) and a small skip between them. Matches the EPUB rendering.
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5\baselineskip plus 0.1\baselineskip}

% Never orphan the first line of a paragraph at the bottom of a page,
% never widow the last line at the top of the next page.
\clubpenalty=10000
\widowpenalty=10000
\raggedbottom

% ----- Code blocks -----------------------------------------------------
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines,breakanywhere,commandchars=\\\{\},fontsize=\small
}

% PDFs build with --no-highlight, so Pandoc does not emit Shaded / Tok
% macros. The Highlighting Verbatim env above handles plain mono code.

% Pandoc-injected helper commands our custom template must define.
% \providecommand so any future Pandoc that starts defining them wins.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{#1}

% ----- Page headers and footers ----------------------------------------
% Default LaTeX book class shows page numbers in inconsistent spots
% (top-right on some, nothing on others). Override with fancyhdr so
% every page has the chapter title centered at the top (auto-wrapping
% to two lines if long) and the page number centered at the bottom.
\usepackage{fancyhdr}
\usepackage{etoolbox}          % for \pretocmd on sectioning commands

\pagestyle{fancy}
\fancyhf{}                     % clear everything
\fancyhead[C]{\small\scshape\nouppercase{\leftmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% 'plain' pagestyle (used by \chapter) should match: centered page
% number at the bottom, no header.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% Allow a long chapter title to break across two lines in the header.
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}

% ----- Chapter titles --------------------------------------------------
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\bfseries\filcenter}
  {\fontsize{14pt}{16pt}\selectfont\textsc{\chaptertitlename\ \thechapter}}
  {12pt}
  {\fontsize{22pt}{26pt}\selectfont}
\titlespacing*{\chapter}{0pt}{20pt}{32pt}

% ----- Silence LaTeX's autogenerated section numbering -----------------
% Each verse-heading already carries a semantic id ('v5.2 — Silence…')
% so the '1.1', '1.2' chapter-auto-numbers in front of them in the TOC
% and the body were pure noise. Drop numbering past chapters entirely.
\setcounter{secnumdepth}{0}

% Same in the TOC: show Chapter + Section entries *without* the
% '1', '1.1' prefixes.
\usepackage{tocloft}
\renewcommand{\cftchapnumwidth}{0pt}
\renewcommand{\cftchappresnum}{}
\renewcommand{\cftchapaftersnum}{}
\renewcommand{\cftsecnumwidth}{0pt}
\renewcommand{\cftsecpresnum}{}

% ----- Multi-page Table of Contents ------------------------------------
% Chapters appear in the TOC with titles, no auto-numbers. Verses are
% *not* listed in the TOC — tocdepth = 0 keeps it to main book headings
% only, which is all a reader needs to navigate a paginated book.
\setcounter{tocdepth}{0}

% ----- Keep verses together --------------------------------------------
% Each ## v-heading becomes \subsection. Demand 10 lines of runway
% below the heading; if less, advance to the next page. Longer reserve
% than before so the closing 'Probability: …' line never slips to the
% next page.
\usepackage{needspace}
\pretocmd{\subsection}{\Needspace*{10\baselineskip}}{}{}

% ----- Include SVG cover and back cover as PDF pages -------------------
% If dist/cover.pdf and dist/back-cover.pdf exist (the build script
% rasterises publish/covers/*.svg to PDF via rsvg-convert), they get
% included as first and last pages so the PDF is a "book with covers"
% out of the box.
\usepackage{pdfpages}

% Some metadata placeholders Pandoc may emit.
$if(title)$
\title{$title$}
$endif$
$if(author)$
\author{$for(author)$$author$$sep$ \and $endfor$}
$endif$
$if(date)$
\date{$date$}
$endif$

\begin{document}

% Front cover. First thing in the PDF, no page number.
$if(cover-pdf)$
\thispagestyle{empty}
\includepdf[pages=1,fitpaper=true]{$cover-pdf$}
$endif$

$if(title)$
\maketitle
$endif$

$if(toc)$
\tableofcontents
\clearpage
$endif$

$body$

% Back cover. Last thing in the PDF.
$if(back-cover-pdf)$
\thispagestyle{empty}
\clearpage
\includepdf[pages=1,fitpaper=true]{$back-cover-pdf$}
$endif$

\end{document}
